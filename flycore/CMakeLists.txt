cmake_minimum_required(VERSION 3.20)
project(flycore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(FC_SOURCES src/flycore.cpp src/sha256.cpp)

# Shared library (Windows editor: flycore.dll; Android: libflycore.so)
add_library(flycore SHARED ${FC_SOURCES})
target_include_directories(flycore PUBLIC include PRIVATE src)
target_compile_definitions(flycore PRIVATE FC_BUILD=1)
set_target_properties(flycore PROPERTIES OUTPUT_NAME "flycore" PREFIX "" C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)
if(ANDROID)
  # 16 KB pages (Google Play) - check with tools/check_16kb.py
  target_link_options(flycore PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
  set_target_properties(flycore PROPERTIES PREFIX "lib")
endif()
if(MSVC)
  target_compile_options(flycore PRIVATE /W4 /permissive- /O2 /fp:precise)
else()
  target_compile_options(flycore PRIVATE -Wall -Wextra -O2 -fno-fast-math -fno-exceptions)
endif()

# Static library for tests and for iOS/Web (Unity compiles the sources directly on those targets)
add_library(flycore_static STATIC ${FC_SOURCES})
target_include_directories(flycore_static PUBLIC include PRIVATE src)
target_compile_definitions(flycore_static PUBLIC FC_STATIC=1 PRIVATE FC_BUILD=1)

option(FC_BUILD_TESTS "Build doctest unit tests" ON)
if(FC_BUILD_TESTS AND NOT ANDROID AND NOT IOS)
  enable_testing()
  add_executable(flycore_tests tests/test_flycore.cpp tests/model_builder.cpp)
  target_include_directories(flycore_tests PRIVATE tests src)
  target_link_libraries(flycore_tests PRIVATE flycore_static)
  add_test(NAME flycore_tests COMMAND flycore_tests)
  # Simple benchmark (not an acceptance test; measures the cost per tick)
  add_executable(flycore_bench tools/bench.cpp tests/model_builder.cpp)
  target_include_directories(flycore_bench PRIVATE tests src)
  target_link_libraries(flycore_bench PRIVATE flycore_static)
  # CLI tool: inspect a .tdfm and run a simple trial (used by research for parity)
  add_executable(flycore_cli tools/cli.cpp)
  target_include_directories(flycore_cli PRIVATE src)
  target_link_libraries(flycore_cli PRIVATE flycore_static)
endif()
